cmake_minimum_required(VERSION 3.20)
project(prismarine_java_math LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED)
if(NOT NODE_INCLUDE_DIR)
  message(FATAL_ERROR "Set NODE_INCLUDE_DIR to the directory containing node_api.h")
endif()
add_library(java_math SHARED java_math.cc)
target_include_directories(java_math PRIVATE ${NODE_INCLUDE_DIR} ${JNI_INCLUDE_DIRS})
set_target_properties(java_math PROPERTIES PREFIX "" SUFFIX ".node")
if(WIN32)
  if(NOT NODE_LIBRARY)
    message(FATAL_ERROR "Set NODE_LIBRARY to node.lib")
  endif()
  target_link_libraries(java_math PRIVATE ${NODE_LIBRARY})
else()
  target_link_libraries(java_math PRIVATE ${CMAKE_DL_LIBS})
  if(APPLE)
    target_link_options(java_math PRIVATE "-undefined" "dynamic_lookup")
  endif()
endif()
